def load_data(C:\Users\Downloads\Va\ecommerce_completed_dataset1):
    return pd.read_csv(C:\Users\Downloads\Va\ecommerce_completed_dataset1)

def clean_data(df):
    df = df.drop_duplicates()
    df = df.fillna(method="ffill")
    return df

def return_rate(df):
    return (df["Returned"].sum() / len(df)) * 100

import matplotlib.pyplot as plt

def plot_returns(df):
    df["Returned"].value_counts().plot(kind="bar")
    plt.title("Returned vs Not Returned")
    plt.show()
